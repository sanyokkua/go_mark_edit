import datetime, json, os, shlex, subprocess, sys, time
from pathlib import Path
base=Path(__file__).resolve().parent
kind=sys.argv[1]
if kind=='go':
    command=['rtk','proxy','go','test','-v','-count=1','-timeout=10s','.']
    cwd=base/'go'
    overrides={'GOCACHE':str(base/'go-cache'),'GOMODCACHE':str(base/'go-modcache'),'GOWORK':'off','GOPROXY':'off','GOSUMDB':'off','GOTOOLCHAIN':'local'}
elif kind=='node':
    command=['rtk','proxy','node','projection_probe.cjs']
    cwd=base/'node'
    overrides={}
else:
    raise SystemExit('unknown probe')
env=os.environ.copy(); env.update(overrides)
exact='cd '+shlex.quote(str(cwd))+'\n'+(' '.join(shlex.quote(k+'='+v) for k,v in overrides.items())+' ' if overrides else '')+shlex.join(command)+'\n'
(base/(kind+'-command.txt')).write_text(exact)
started=time.monotonic()
record={'command':command,'cwd':str(cwd),'environment_overrides':overrides,'external_timeout_seconds':30,'utc_started':datetime.datetime.now(datetime.timezone.utc).isoformat()}
try:
    result=subprocess.run(command,cwd=cwd,env=env,capture_output=True,timeout=30)
    stdout=result.stdout; stderr=result.stderr
    record.update(exit_code=result.returncode,timed_out=False)
except subprocess.TimeoutExpired as error:
    stdout=error.stdout or b''; stderr=error.stderr or b''
    record.update(exit_code=None,timed_out=True)
record['elapsed_seconds']=round(time.monotonic()-started,3)
(base/(kind+'-stdout.log')).write_bytes(stdout)
(base/(kind+'-stderr.log')).write_bytes(stderr)
(base/(kind+'-result.json')).write_text(json.dumps(record,indent=2)+'\n')
print(json.dumps(record,indent=2))
print(stdout.decode(errors='replace'),end='')
if stderr: print('STDERR:\n'+stderr.decode(errors='replace'),end='')
