package application

import (
    "context"
    "testing"
)

func TestLostCloseRequestRequiresExplicitReset(t *testing.T) {
    ctx := context.Background()
    emissions, deliveries, quitCalls := 0, 0, 0
    var receiver func()
    var coordinator *CloseCoordinator
    permitVeto := true
    coordinator = NewCloseCoordinator(func(context.Context) {
        emissions++
        if receiver != nil { receiver() }
    }, func(context.Context) {
        quitCalls++
        permitVeto = coordinator.BeforeClose(ctx)
    })

    firstVeto := coordinator.BeforeClose(ctx)
    t.Logf("no responder: veto=%t emissions=%d deliveries=%d quitCalls=%d", firstVeto, emissions, deliveries, quitCalls)
    if !firstVeto || emissions != 1 || deliveries != 0 || quitCalls != 0 {
        t.Fatal("first request must be vetoed and emitted once without a delivery")
    }

    receiver = func() { deliveries++ }
    repeatedVeto := coordinator.BeforeClose(ctx)
    t.Logf("receiver restored, repeat close: veto=%t emissions=%d deliveries=%d quitCalls=%d", repeatedVeto, emissions, deliveries, quitCalls)
    if !repeatedVeto || emissions != 1 || deliveries != 0 || quitCalls != 0 {
        t.Fatal("pending request should demonstrate missing replay after receiver restoration")
    }

    coordinator.Cancel()
    resetVeto := coordinator.BeforeClose(ctx)
    t.Logf("Cancel then close: veto=%t emissions=%d deliveries=%d quitCalls=%d", resetVeto, emissions, deliveries, quitCalls)
    if !resetVeto || emissions != 2 || deliveries != 1 || quitCalls != 0 {
        t.Fatal("positive control: Cancel must allow a fresh delivered request")
    }

    if err := coordinator.Authorize(ctx); err != nil { t.Fatalf("Authorize: %v", err) }
    t.Logf("Authorize: native callback veto=%t emissions=%d deliveries=%d quitCalls=%d", permitVeto, emissions, deliveries, quitCalls)
    if permitVeto || emissions != 2 || deliveries != 1 || quitCalls != 1 {
        t.Fatal("positive control: authorization must create and consume a permit")
    }
}
