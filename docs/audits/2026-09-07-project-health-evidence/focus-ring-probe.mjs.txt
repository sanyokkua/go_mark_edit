async function inspectMenu(page, label) {
  return page.evaluate((label) => {
    const el = document.activeElement;
    const menu = document.querySelector('[role="menu"]');
    const cs = menu ? getComputedStyle(menu) : null;
    return {
      label,
      active: el ? `${el.tagName}.${el.className}`.slice(0, 80) : null,
      menuFound: !!menu,
      menuIsActive: menu === el,
      menuMatchesFocusVisible: menu ? menu.matches(':focus-visible') : null,
      boxShadow: cs ? cs.boxShadow.slice(0, 120) : null,
    };
  }, label);
}
