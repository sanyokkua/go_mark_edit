import hashlib
import json
import os
from pathlib import Path
import signal
import subprocess
import time
import urllib.request

root = Path('/tmp/gomark-navigation-probe.Ig0rZh')
repo = Path('/Users/ok/Development/GitHub/go_mark_edit')
module = Path('/Users/ok/go/pkg/mod/github.com/wailsapp/wails/v2@v2.15.0')
metadata = {'startedAt': time.strftime('%Y-%m-%dT%H:%M:%SZ', time.gmtime()), 'scope': 'HTTP pipeline; no native app or user profile'}
for folder in ['gocache', 'gotmp', 'vite-cache']:
    (root / folder).mkdir(exist_ok=True)
tracked_before = subprocess.check_output(['rtk','proxy','git','diff','--binary'],cwd=repo)
metadata['gitDiffSHA256Before'] = hashlib.sha256(tracked_before).hexdigest()
metadata['head'] = subprocess.check_output(['rtk','proxy','git','rev-parse','HEAD'],cwd=repo,text=True).strip()
inputs = [root/'probe.go', root/'vite.config.mjs', root/'run.py', repo/'frontend/vite.config.ts', repo/'frontend/index.html', repo/'frontend/package-lock.json', repo/'go.mod', repo/'go.sum']
inputs += [module / path for path in ['pkg/assetserver/assetserver.go','pkg/assetserver/assethandler.go','pkg/assetserver/assethandler_external.go','pkg/assetserver/assetserver_dev.go','internal/frontend/runtime/ipc.js','internal/frontend/runtime/ipc_websocket.js','internal/frontend/runtime/runtime_debug_desktop.js']]
inputs += [repo/'frontend/node_modules/vite/dist/node/chunks/config.js']
metadata['inputsSHA256'] = {str(path):hashlib.sha256(path.read_bytes()).hexdigest() for path in inputs}
vite_cmd = ['rtk','proxy','npm','--prefix',str(repo/'frontend'),'run','dev','--','--mode','wails','--config',str(root/'vite.config.mjs'),'--configLoader','runner']
go_cmd = ['rtk','proxy','go','run','-tags','dev',str(root/'probe.go')]
metadata['viteCommand'] = vite_cmd
metadata['probeCommand'] = go_cmd
environment = dict(os.environ)
environment.update({'GOCACHE':str(root/'gocache'),'GOTMPDIR':str(root/'gotmp'),'GOFLAGS':'-mod=readonly','GOPROXY':'off','GOSUMDB':'off','npm_config_cache':str(root/'npm-cache')})
metadata['environmentOverrides'] = {key:environment[key] for key in ['GOCACHE','GOTMPDIR','GOFLAGS','GOPROXY','GOSUMDB','npm_config_cache']}
vite = None
probe = None
exit_code = 1
try:
    with (root/'vite-output.txt').open('wb') as vite_log, (root/'probe-output.txt').open('wb') as probe_log:
        vite = subprocess.Popen(vite_cmd,cwd=repo,env=environment,stdout=vite_log,stderr=subprocess.STDOUT,start_new_session=True)
        metadata['viteProcessGroup'] = vite.pid
        for attempt in range(80):
            if vite.poll() is not None:
                raise RuntimeError('Task-owned Vite exited before readiness')
            try:
                with urllib.request.urlopen('http://127.0.0.1:54173/',timeout=0.5) as response:
                    if response.status == 200:
                        metadata['viteReady'] = True
                        break
            except Exception:
                time.sleep(0.25)
        else:
            raise RuntimeError('Task-owned Vite readiness deadline expired')
        probe = subprocess.Popen(go_cmd,cwd=repo,env=environment,stdout=probe_log,stderr=subprocess.STDOUT,start_new_session=True)
        metadata['probeProcessGroup'] = probe.pid
        try:
            exit_code = probe.wait(timeout=180)
        except subprocess.TimeoutExpired:
            metadata['probeTimeoutSeconds'] = 180
            os.killpg(probe.pid,signal.SIGTERM)
            try:
                probe.wait(timeout=5)
            except subprocess.TimeoutExpired:
                os.killpg(probe.pid,signal.SIGKILL)
                probe.wait()
            exit_code = 124
        metadata['probeExitCode'] = exit_code
except Exception as error:
    metadata['error'] = repr(error)
finally:
    if vite is not None:
        if vite.poll() is None:
            os.killpg(vite.pid,signal.SIGTERM)
            try:
                vite.wait(timeout=5)
            except subprocess.TimeoutExpired:
                os.killpg(vite.pid,signal.SIGKILL)
                vite.wait()
        metadata['viteExitCodeAfterTaskOwnedCleanup'] = vite.returncode
    tracked_after = subprocess.check_output(['rtk','proxy','git','diff','--binary'],cwd=repo)
    metadata['gitDiffSHA256After'] = hashlib.sha256(tracked_after).hexdigest()
    metadata['trackedDiffUnchanged'] = tracked_after == tracked_before
    metadata['gitStatusAfter'] = subprocess.check_output(['rtk','proxy','git','status','--short'],cwd=repo,text=True)
    metadata['completedAt'] = time.strftime('%Y-%m-%dT%H:%M:%SZ',time.gmtime())
    (root/'run-metadata.txt').write_text(json.dumps(metadata,indent=2)+'\n')
print(json.dumps(metadata,indent=2))
raise SystemExit(exit_code)
